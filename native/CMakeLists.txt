cmake_minimum_required(VERSION 3.22.1)
project("aoas")

# AOAS's native library: the USB driver, the shared ring, the ownership state
# machine and the Binder service. Everything except the three Android APIs that
# only exist in Java (see native/jni_bridge.cc for which and why).

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# --- audio_engine -------------------------------------------------------------
#
# The ONLY first-party dependency, and only one target out of it: ae_usb, which
# is UsbAudioDriver plus the vendored libusb. Not the monolithic `matrix_audio`
# library under audio_engine/platform/android -- that one also pulls in libFLAC,
# libmpg123, LAME, AMediaCodec and AAudio, and AOAS decodes nothing and plays to
# nothing but the DAC. There is exactly one implementation of "talk to a USB
# Audio Class device" and this links it rather than copying it.
#
# Path override exists because this repo is expected to move audio_engine to a
# git submodule under framework/ (matching app_shell and vk_canvas); until then
# it sits in firstparty/.
set(AOAS_AUDIO_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../firstparty/audio_engine"
    CACHE PATH "Path to the audio_engine checkout")
if(NOT EXISTS "${AOAS_AUDIO_ENGINE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR
        "audio_engine not found at ${AOAS_AUDIO_ENGINE_DIR}. "
        "It carries the USB Audio driver AOAS is built on; without it there is "
        "nothing to build. Set -DAOAS_AUDIO_ENGINE_DIR=<path>.")
endif()

# No decoders: AOAS never opens a file. Turning them off also means the FLAC and
# mpg123 dependencies are not needed at all, so a shallow clone still builds.
set(AUDIO_ENGINE_BUILD_DECODERS OFF CACHE BOOL "" FORCE)
set(AUDIO_ENGINE_BUILD_TOOLS    OFF CACHE BOOL "" FORCE)
add_subdirectory(${AOAS_AUDIO_ENGINE_DIR} ${CMAKE_BINARY_DIR}/audio_engine)

# --- the UI frameworks --------------------------------------------------------
#
# AOAS's console is C++ drawn with vk_canvas, hosted by app_shell's
# NativeActivity Host. There is no Java Activity and no layout XML in this app;
# see native/ui/android_main.cc.
#
# app_shell's own CLAUDE.md was once read here as "app_shell cannot be used" --
# AndroidHost::init blocks waiting for a window, which a headless service will
# never get. That is true and irrelevant: the console is an ACTIVITY, and it
# gets a window like any other. Only the service half runs windowless, and it
# links none of this.
set(VK_CANVAS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../firstparty/Vk_Canvas_Lb_LAW
    CACHE PATH "" FORCE)
set(APP_SHELL_DIR_LOCAL ${CMAKE_CURRENT_SOURCE_DIR}/../firstparty/App_shell)
set(APP_SHELL_APP_NAME "aoas" CACHE STRING "" FORCE)

# native_app_glue ships inside the NDK, so only a consumer knows where it is.
set(NATIVE_APP_GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
set(APP_SHELL_NATIVE_APP_GLUE_DIR ${NATIVE_APP_GLUE} CACHE PATH "" FORCE)

add_subdirectory(${VK_CANVAS_DIR}/core     ${CMAKE_BINARY_DIR}/vk_canvas_core)
add_subdirectory(${APP_SHELL_DIR_LOCAL}    ${CMAKE_BINARY_DIR}/app_shell)

# The MSDF legibility floor is produced by a tool that has to RUN on the build
# machine, so this cross build cannot make it. The desktop build does; this one
# reads it out of that tree, and says so plainly rather than failing later with
# a missing header.
set(AOAS_GENERATED_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../build/desktop/generated
    CACHE PATH "Where a desktop configure left ui_min_text_size.gen.h")
if(NOT EXISTS ${AOAS_GENERATED_DIR}/ui_min_text_size.gen.h)
    message(FATAL_ERROR
        "ui_min_text_size.gen.h is missing. It is generated by a host tool, so "
        "the Android build cannot make it: configure the desktop build first "
        "(cmake -S . -B build/desktop), or point -DAOAS_GENERATED_DIR at a tree "
        "that has it.")
endif()
# app_shell's ui_metrics.hh includes it and looks in ${CMAKE_BINARY_DIR}/generated,
# so the copy has to land there -- an include path on our own target would not
# help the library doing the including.
configure_file(${AOAS_GENERATED_DIR}/ui_min_text_size.gen.h
               ${CMAKE_BINARY_DIR}/generated/ui_min_text_size.gen.h COPYONLY)

# --- the library --------------------------------------------------------------
#
# No AIDL sources here: the interface is generated with the Java backend by
# Gradle (see aoas_service.hh for why that beats the NDK backend), and the
# generated stub calls down through jni_bridge.cc.
# One library, one process. The console is a NativeActivity in AOAS's own
# process, exactly where the Java one ran, so it reads aoas::processServer()
# directly instead of going back out through Binder to reach the object next to
# it. Splitting the UI into its own process is a real and separate improvement
# -- a Vulkan frame cannot then contend with the thread feeding the isochronous
# endpoint -- but it is not what this change is, and doing both at once would
# have made neither reviewable.
add_library(aoas SHARED
    aoas_service.cc
    jni_bridge.cc
    relay.cc
    usb_device.cc
    ui/android_main.cc
    ui/console_app.cc
    ui/console_draw.cc
    ui/service_source.cc
    ${NATIVE_APP_GLUE}/android_native_app_glue.c
    # vk_canvas's Android platform seam: AndroidSurfaceProvider and
    # AndroidAssetReader, which AndroidHost constructs. A .cc a consumer
    # compiles rather than a library, the same way the Win32 backend is.
    ${VK_CANVAS_DIR}/platform/android/android_platform.cc)

target_include_directories(aoas PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/ui
    ${NATIVE_APP_GLUE}
    ${VK_CANVAS_DIR}/platform/android)   # jni_util.hh

# -Wconversion stays off the framework sources: they are first-party but not
# this project's, and turning their warnings into this build's noise would
# train everyone here to ignore the ones that are ours.
target_compile_options(aoas PRIVATE -O2 -Wall -Wextra)

target_link_libraries(aoas
    ae_usb              # UsbAudioDriver + libusb (first-party)
    app_shell_android   # the NativeActivity Host
    vk_canvas_core      # the renderer the console draws through
    android             # ASharedMemory_create for the cross-process ring
    log
    vulkan)

# The glue's entry point has no caller in this translation unit; without this
# the linker garbage-collects the symbol NativeActivity looks up by name.
target_link_options(aoas PRIVATE -u ANativeActivity_onCreate)
